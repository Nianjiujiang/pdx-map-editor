
/* ================================================================ 手机版界面
 *
 * 只打进手机版，电脑版源码不受影响。
 *
 * 一级：地图 + 顶栏（左 返回 · 正中 永夜廿九 · 右 教程 / 导入导出）
 *             + 底部四个入口（显示设置 · 颜色 · 工具 · 视图）
 * 二级：从所按按钮正上方弹出，带一个指向按钮的小尖角；
 *       顶栏的「导入导出」按钮在上方，因此那一栏向下弹；
 *       「设置」的内容也在这一层，不再套第三层。
 *
 * 面板内容直接搬运桌面现成节点，功能、状态、事件全部沿用。
 * 搬运逐步单独兜错，任一步失败不影响入口按钮。
 */
function bindMobile() {
  const stage = $('stage');
  const app = $('app');
  if (!stage || !app) return;

  /* ============================================================ 一级界面 */
  const ui = document.createElement('div');
  ui.id = 'mob-ui';
  ui.innerHTML =
    '<div id="mob-backdrop"></div>'
    + '<div id="mob-topbar">'
    +   '<button type="button" class="mob-btn mob-back-arrow" id="mob-back" aria-label="回主菜单">\u2190</button>'
    +   '<span id="mob-mid"></span>'
    +   '<button type="button" class="mob-btn" id="mob-io">文件</button>'
    + '</div>'
    + '<div id="mob-dock">'
    +   '<button type="button" data-sheet="display"><i>\u2630</i>设置</button>'
    +   '<button type="button" data-sheet="color"><i>\u25c9</i>颜色</button>'
    +   '<button type="button" data-sheet="tool" id="mob-dock-tool"><i>\u270e</i>工具</button>'
    +   '<button type="button" data-sheet="view"><i>\u25a6</i>视图</button>'
    + '</div>'
    + '<div id="mob-sheet">'
    +   '<div id="mob-sheet-head"><b id="mob-sheet-title">面板</b><span></span>'
    +     '<button type="button" id="mob-sheet-x">收起</button></div>'
    +   '<div id="mob-sheet-body"></div>'
    + '</div>';
  app.appendChild(ui);

  const sheet = $('mob-sheet');
  const body = $('mob-sheet-body');
  const backdrop = $('mob-backdrop');
  const titleEl = $('mob-sheet-title');
  const wait = (fn) => { try { fn(); } catch (e) { if (window.console) console.warn('[手机版]', e && e.message); } };

  /* ============================================================ 选文件：过滤整个放开
   * .pdxmap 是自定义扩展名，系统上报给选择器的类型是 application/octet-stream；
   * 安卓 / iOS 的文件选择器按 MIME 过滤，会把该文件判成不可导入。
   * 这里对手机一律放开 accept（源头已补 octet-stream，此处再兜一道）；
   * 选错的文件由 pack.js 的校验器给出具体提示。
   */
  wait(() => {
    const pf = $('pack-file');
    if (pf && pf.accept !== '*/*') pf.accept = '*/*';
  });

  /* ============================================================ 面板骨架（先造好）
   * 先造骨架并接好按钮监听，再搬节点：某一步搬运失败时，点入口仍能出面板。
   */
  const group = {};
  const makeGroup = (key, title) => {
    const g = document.createElement('div');
    g.className = 'mob-group';
    g.setAttribute('data-group', key);
    g.hidden = true;
    body.appendChild(g);
    group[key] = { el: g, title: title };
    return g;
  };
  const gColor = makeGroup('color', '颜色');
  const gTool = makeGroup('tool', '工具');
  const gView = makeGroup('view', '视图');
  const gDisp = makeGroup('display', '设置');
  const gIO = makeGroup('io', '文件');

  /* 🎨 颜色页小窗（调色板 / 最近使用 / 色标）的高度 = 面板剩余空间。
   *   不用 46vh：vh 按整屏算，而面板只占屏幕一部分，小窗会比面板还高，
   *   最后一排被裁在面板外（mobile_css.txt 里那条 vh 只是 JS 未运行时的兜底）。
   *   量的是它在分组里占掉多少，需加回 scrollTop，否则滚过一屏后越量越矮。
   *   分组隐藏或尚未排版时不量（结果会是 0 或负数）。
   *   底部还有 padding-bottom 与安全区，需留出余量。 */
  const fitColorKit = () => {
    try {
      const g = group.color && group.color.el;
      if (!g) return;
      const kit = g.querySelector('.mob-color-kit');
      if (!kit) return;
      if (g.hidden) { kit.style.maxHeight = ''; return; }
      const gr = g.getBoundingClientRect();
      if (!(gr.height > 40)) return;               // 面板尚未撑开：下一帧或下次 place 再量
      const kr = kit.getBoundingClientRect();
      const topIn = kr.top - gr.top + (g.scrollTop || 0);
      const room = Math.round(gr.height - topIn - 12);
      kit.style.maxHeight = (room > 150 ? room : 150) + 'px';
    } catch (e) { /* 量不出来时由 CSS 兜底 */ }
  };

  /** 把面板摆到按钮正上方；按钮在上半屏时改为向下弹 */
  const place = (btn, key) => {
    try {
      const a = app.getBoundingClientRect();
      const r = btn.getBoundingClientRect();
      /* 每一栏按自己的内容定宽：
       *   · 工具 / 视图 / 文件：内容短，取窄宽
       *   · 设置 / 颜色：成排的开关和输入框，摊到接近整屏（a.width - 16）
       */
      const WANT = { tool: 148, view: 148, io: 148 };
      const FULL = { display: 1, color: 1 };
      const w = FULL[key] ? (a.width - 16) : Math.min(a.width - 16, WANT[key] || 300);
      const cx = r.left - a.left + r.width / 2;
      let left = cx - w / 2;
      left = Math.max(8, Math.min(a.width - w - 8, left));
      const below = (r.top - a.top) < a.height / 2;
      sheet.style.width = Math.round(w) + 'px';
      sheet.style.left = Math.round(left) + 'px';
      sheet.classList.toggle('below', below);
      /* 高度系数：两次「减去三成」叠加后约为 0.5（0.7 × 0.7）。
       *   计算对象是「从按钮到屏幕边缘」的可用高度，结果写成内联样式，
       *   优先级高于 mobile_css.txt 中的 #mob-sheet { max-height: … }，
       *   所以只改 CSS 不会生效，系数必须在这里乘。
       *   剩余空间留给地图，放不下的内容交给 #mob-sheet-body 的滑条。
       *   调整系数时需同步修改 CSS 里那两处 vh 兜底值，避免两边相差过大。 */
      const _avail = (below ? (a.height - (r.bottom - a.top) - 20) : (r.top - a.top - 20));
      /* 「更高」模式：上限回到可用高度本身，不再乘系数。
       *   这是「小按钮没有效果」的真正原因：把上限压到 0.5 倍可用高度后，
       *   面板的 height 与 max-height 都取自同一个上限，
       *   按不按「更高」得到的是同一个数值。该按钮的语义是面板整体变高，
       *   不是把内容区撑开。
       *   普通模式 = 可用高度 × 0.5；「更高」= 可用高度 × 1.0（顶到屏幕边但不溢出）。
       *   系数只在这一处计算，别处不得重复相乘。 */
      const _tall = sheet.classList.contains('mob-tall');
      /* 系数按页区分：上面那些系数（0.7 / 0.5 / 0.333 / 0.4）都是为设置页调的，
       *   但它是全局值，颜色页会跟着一起缩到四成；颜色页里是调色板与图例列表，
       *   本应占地方。
       *   分开算：display（设置）→ 0.46（= 0.4 × 1.15）；
       *   color（颜色）与其余页（工具 / 视图 / 文件）→ 1.0，按内容撑开。
       *   高度由下面的内联 sheet.style.maxHeight 决定，CSS 里那几条被它压过；
       *   判断用 key（place(btn, key) 的入参），不要猜当前是哪一页。 */
      const _isDisplay = (key === 'display');
      /* 设置页那个 0.46 = 0.4 × 1.15，改这里才生效，CSS 里那两条被内联样式压过。
       *
       * 高度只有一个来源：三处若同时写高度会互相争夺 ——
       *   ① 这里按系数算出的 sheet.style.maxHeight
       *   ② 下面把 height 定成 maxHeight
       *   ③ mobile_css.txt 的 #mob-sheet.mob-tall { height: … !important }
       * ③ 的 !important 压掉了 ①②，导致「更高」无视屏幕尺寸固定为一个 vh 值。
       * 现已去掉 ③ 的 !important（CSS 只留兜底）：
       *   · 普通模式不写 height，面板按内容撑开；
       *   · 「更高」时 height = maxHeight，多出的内容进滑条。
       * 调整高度只改这一处，不要再往 CSS 里加大数字。 */
      const _f = _tall ? 1.0 : (_isDisplay ? 0.46 : 1.0);
      const _cap = Math.round(Math.max(160, _avail) * _f) + 'px';
      if (below) {
        sheet.style.top = Math.round(r.bottom - a.top + 10) + 'px';
        sheet.style.bottom = 'auto';
        sheet.style.maxHeight = _cap;
      } else {
        sheet.style.bottom = Math.round(a.bottom - r.top + 10) + 'px';
        sheet.style.top = 'auto';
        sheet.style.maxHeight = _cap;
      }
      sheet.style.setProperty('--tail-x', Math.round(cx - left) + 'px');
      /* 「更高」模式：把 height 顶到刚算出的上限（贴着底部往上长，不顶出屏幕）；
       *   普通模式清掉 height，回到按内容撑开。
       *   这里不做动画：显式 px 会让面板不再随内容长高。
       *   ⚠ 这一层不再乘系数 —— maxHeight 已按系数缩过，再乘一遍会矮一倍。
       *   ⚠ maxHeight 是字符串（形如 512px），要 parseFloat 再算。 */
      try {
        if (sheet.classList.contains('mob-tall') && sheet.style.maxHeight) {
          /* 系数 1.0：maxHeight 上面已经乘过 */
          sheet.style.height = sheet.style.maxHeight;
        } else {
          sheet.style.height = '';
        }
      } catch (e) { /* 忽略 */ }
      /* 颜色页小窗跟随面板高度：开「更高」、转屏、改窗口大小都会重走 place */
      if (key === 'color') {
        fitColorKit();
        try { requestAnimationFrame(() => { try { fitColorKit(); } catch (e) { /* 忽略 */ } }); } catch (e) { /* 忽略 */ }
      }
    } catch (e) { /* 算不出来时按默认位置 */ }
  };

  let opened = null, openedBtn = null;
  const setSheet = (key, btn) => {
    const g = key ? group[key] : null;
    for (const k in group) group[k].el.hidden = !(g && group[k] === g);
    if (g) {
      titleEl.textContent = g.title;
      /* 「更高」按页各记各的：打开哪一页就套哪一页的状态。
       * ⚠ 这一句必须在 place() 之前：
       *   ① 少了它，按钮加上的 mob-tall 没人摘，之后打开工具 / 视图 / 文件也全变高
       *      （整个 App 只有一个 #mob-sheet，四个入口共用）；
       *   ② place() 看着这个 class 决定高度，顺序反了会按上一页的状态算一次。
       */
      try {
        const _ts = (group.display && group.display.el && group.display.el._tallState) || {};
        sheet.classList.toggle('mob-tall', !!_ts[key]);
      } catch (e) { /* 忽略 */ }
      if (btn) place(btn, key);
      // 高度不锁，跟「颜色」那栏一样按内容撑开；设置页例外：下面用 _lockSoon 量出最高页锁住，切页不跳。
      sheet.classList.remove('tall');
      if (key === 'display' && group.display && group.display.el._lockSoon) {
        requestAnimationFrame(() => { try { group.display.el._lockSoon(); } catch (e) { /* 忽略 */ } });
      }
      // 横向滚动位置归零，否则内容稍宽就会被推到一边，左边一截看不见
      try { body.scrollLeft = 0; sheet.scrollLeft = 0; } catch (e) { /* 忽略 */ }
      /* 打开设置面板时，若正停在「导出图例」那一页，把列表重刷一遍。
       * 用 legendFresh：它会先按当前剧本 / 层级把色块补算一遍再列，
       * 只刷列表的话，列出来的可能还是上一个剧本的分组。 */
      if (key === 'display') {
        const _dEl = group.display && group.display.el ? group.display.el : null;
        const _lgPane = _dEl ? _dEl.querySelector('.mob-pane[data-pane="legend"]') : null;
        if (_lgPane && !_lgPane.hidden) {
          try { legendFresh(); } catch (e) { /* 忽略 */ }
        }
      }
      sheet.classList.add('on');
      backdrop.classList.add('on');
      opened = key;
      openedBtn = btn || null;
      /* 颜色页小窗要等面板真正显出来（有高度）才量得到，place() 里那次可能太早 */
      if (key === 'color') {
        try { requestAnimationFrame(() => { try { fitColorKit(); } catch (e) { /* 忽略 */ } }); } catch (e) { /* 忽略 */ }
      }
    } else {
      sheet.classList.remove('on');
      backdrop.classList.remove('on');
      opened = null;
      openedBtn = null;
      /* 收起时清掉量出的高度，否则下次开面板会先按上一屏的高度闪一下 */
      try {
        const k = group.color && group.color.el && group.color.el.querySelector('.mob-color-kit');
        if (k) k.style.maxHeight = '';
      } catch (e) { /* 忽略 */ }
    }
    const bs = ui.querySelectorAll('#mob-dock button');
    for (let i = 0; i < bs.length; i++) bs[i].classList.toggle('on', bs[i].getAttribute('data-sheet') === key);
    const io = $('mob-io');
    if (io) io.classList.toggle('on', key === 'io');
  };

  const dockWs = ui.querySelectorAll('#mob-dock button');
  for (let i = 0; i < dockWs.length; i++) {
    (function (b) {
      b.addEventListener('click', () => {
        const k = b.getAttribute('data-sheet');
        setSheet(opened === k ? null : k, b);
      });
    })(dockWs[i]);
  }
  const ioBtn = $('mob-io');
  if (ioBtn) ioBtn.addEventListener('click', () => setSheet(opened === 'io' ? null : 'io', ioBtn));
  const backBtn = $('mob-back');
  if (backBtn) backBtn.addEventListener('click', () => {
    // 只做一件事：回主菜单，不兼职关弹窗、收面板。
    // 不允许丢失已绘内容：先写入暂存区，回来选同一张图会自动接上（暂存按地图分开存）。
    let dirty = 0;
    try { dirty = (state.changed && state.changed.size) || 0; } catch (e) { dirty = 0; }
    if (dirty > 0 && !stashNow()) {
      // 暂存区不可用（无痕模式 / 配额满）时才提示，避免静默丢弃
      const ok = window.confirm('这台浏览器不让暂存，回主菜单会丢掉还没导出的涂色（改动 ' + dirty + ' 处）。\n'
        + '想留着就先点「文件 → 存入地图仓库」存一份。\n\n确定要回主菜单吗？');
      if (!ok) return;
    }
    // 换地图需要重建数据与渲染器，页面自带的启动流程正好做这件事；
    // 重载后同一张图的涂色由 loadProject() 自动接上。
    location.reload();
  });
  if ($('mob-sheet-x')) $('mob-sheet-x').addEventListener('click', () => setSheet(null));
  if (backdrop) backdrop.addEventListener('click', () => setSheet(null));
  document.addEventListener('keydown', (e) => { if (e.key === 'Escape') setSheet(null); });
  // 横竖屏切换时重新摆放面板
  window.addEventListener('resize', () => { if (opened && openedBtn) place(openedBtn, opened); });

  /* ============================================================ 再搬节点 */
  const panelOf = (sel) => {
    const n = typeof sel === 'string' ? $(sel) : sel;
    return n && n.closest ? n.closest('section.panel') : null;
  };
  const move = (g, node) => { if (node && node.parentNode !== g) g.appendChild(node); };
  const movePanelOf = (g, sel) => { move(g, panelOf(sel)); };

  wait(() => {
    movePanelOf(gColor, 'brush-hex');        // 当前颜色：色块 / HEX / RGB / 标记
    movePanelOf(gColor, 'results');          // 搜索
    move(gColor, $('label-pick'));

    /* ============================================================ 颜色：取消二级菜单
     * 手机版点大色块原本弹出 #brush-menu（自定义颜色 / 收藏 / 最近使用），
     * 它是浮层，会被面板裁掉；而点色块的本意是改颜色。
     * 处理方式：① 点大色块 = 打开系统取色器（走 #brush-menu-custom 那条路）；
     *       ② 收藏调色板 / 最近使用搬到面板底部，常驻可见；
     *       ③ 色块长按 = 删除（手机没有右键）。
     * 只影响手机版，桌面版那颗色块仍走二级菜单。
     */
    const colorPanel = panelOf($('brush-swatch')) || gColor;
    const bmenu = $('brush-menu');
    const fav = $('palette-fav');
    const recent = $('palette');
    /* 删色的结果提示：长按回调在 if 之外，先留空壳，实现见下面的 if */
    let sayHint = null;
    if (colorPanel && bmenu && fav && recent) {
      /* 把二级菜单里那两段（标题 + 色块）整段搬出来。
         ⚠ 标题节点是 palette 的前一个兄弟，不能只搬色块，
           否则面板底下会出现两排没有名字的方块。 */
      const kids = Array.prototype.slice.call(bmenu.children || []);
      const favCap = kids[kids.indexOf(fav) - 1] || null;
      const recentCap = kids[kids.indexOf(recent) - 1] || null;
      const kit = document.createElement('div');
      kit.className = 'mob-color-kit';
      if (favCap) kit.appendChild(favCap);
      kit.appendChild(fav);
      if (recentCap) kit.appendChild(recentCap);
      kit.appendChild(recent);
      /* 这行小字平时不显示，只用于报告删除结果
         （全局浮条 TOAST_OFF 关闭时 toast() 是空操作）。 */
      const hint = document.createElement('div');
      hint.className = 'mob-color-hint';
      hint.hidden = true;
      kit.appendChild(hint);
      let hintTimer = 0;
      sayHint = (msg) => {
        clearTimeout(hintTimer);
        hint.textContent = msg || '';
        hint.hidden = !msg;
        hintTimer = setTimeout(() => { hint.textContent = ''; hint.hidden = true; }, 2400);
      };
      /* 🎨 色标（ColorBrewer）：跟桌面同一套数据，一行八个 26px 格子 */
      try {
        if (window.__PDX_BREWER) {
          const brwCap = document.createElement('div');
          brwCap.className = 'brush-menu-cap';   // 与「调色板 / 最近使用」同一套标题样式
          brwCap.textContent = '色标';
          const brw = document.createElement('div');
          brw.className = 'mob-brw';
          window.__PDX_BREWER.buildInto(brw);
          brw.addEventListener('click', (ev) => {
            const t = ev.target;
            const hex = t && t.dataset ? t.dataset.c : null;
            if (hex) window.__PDX_BREWER.setColor(hex);
          });
          kit.appendChild(brwCap);
          kit.appendChild(brw);
        }
      } catch (e) { /* 色标没装上不影响别的 */ }
      colorPanel.appendChild(kit);
      /* 二级菜单整体停用：自定义颜色按钮保留但不显示，
         大色块的 onclick 会直接调用它。 */
      bmenu.hidden = true;
      bmenu.style.display = 'none';
    }

    /* 大色块 = 改颜色（不再弹二级菜单）*/
    const sw = $('brush-swatch');
    if (sw) sw.onclick = () => {
      const custom = $('brush-menu-custom');
      if (custom && typeof custom.onclick === 'function') custom.onclick();
    };

    /* 长按色块 = 删除（手机没有右键，与地图上长按取色是同一个手势）
     *  · 收藏调色板：长按 → paletteRemoveAt(i)
     *  · 最近使用：长按 → removeRecent(rgb)
     *  · 「＋」那个收色块不删
     *  ⚠ 长按之后浏览器会补发一个 click，用 suppress 吃掉，
     *    否则删完会顺手把颜色设成画笔色。
     */
    const bindLongDelete = (box, onDelete) => {
      if (!box) return;
      let timer = 0, sx = 0, sy = 0, suppress = false;
      const cancel = () => { clearTimeout(timer); timer = 0; };
      const chipHit = (target) => {
        const chip = target && target.closest ? target.closest('.chip') : null;
        if (!chip || (chip.classList && chip.classList.contains('palette-plus'))) return null;
        const chips = Array.prototype.slice.call(box.children || [])
          .filter((x) => x.classList && x.classList.contains('chip')
            && !x.classList.contains('palette-plus'));
        const i = chips.indexOf(chip);
        return i >= 0 ? { chip: chip, i: i } : null;
      };
      box.addEventListener('touchstart', (e) => {
        const hit = chipHit(e.target);
        if (!hit) return;
        suppress = false;              // 新手势开始，上次长按的 click 抑制作废
        const t = e.touches && e.touches[0];
        sx = t ? t.clientX : 0; sy = t ? t.clientY : 0;
        cancel();
        timer = setTimeout(() => {
          timer = 0;
          suppress = true;
          /* 这 0.5 秒内色块可能已被右键 / contextmenu 删掉，再删一次会误伤下一个，
             先确认它仍在原位置 */
          if (Array.prototype.indexOf.call(box.children || [], hit.chip) < 0) return;
          try { onDelete(hit.i); } catch (err) { /* 删除失败不影响手势 */ }
          try { if (navigator && navigator.vibrate) navigator.vibrate(20); } catch (err) { /* 忽略 */ }
        }, 520);
      }, { passive: true });
      box.addEventListener('touchmove', (e) => {
        const t = e.touches && e.touches[0];
        if (!t) return cancel();
        if (Math.abs(t.clientX - sx) > 10 || Math.abs(t.clientY - sy) > 10) cancel();
      }, { passive: true });
      box.addEventListener('touchend', cancel);
      box.addEventListener('touchcancel', cancel);
      /* 桌面右键走 contextmenu：先取消长按计时，避免重复删除。
         ⚠ 长按已经删过时，不再让 chip 自己的 contextmenu 再删一次，
           否则删掉的是下一个颜色。 */
      box.addEventListener('contextmenu', (e) => {
        cancel();
        if (suppress && e) {
          if (e.preventDefault) e.preventDefault();
          if (e.stopPropagation) e.stopPropagation();
        }
      }, true);
      box.addEventListener('click', (e) => {
        if (!suppress) return;
        suppress = false;
        if (e && e.preventDefault) e.preventDefault();
        if (e && e.stopPropagation) e.stopPropagation();
      }, true);
    };

      const hex = (typeof paletteList === 'function' ? paletteList()[i] : '') || '';
      try { paletteRemoveAt(i); } catch (e) { /* 忽略 */ }

      try {

      } catch (e) { /* 忽略 */ }
    });

      { key: 'legend', title: '图例' },

    gDisp._lockHMax = 0;

         *   面板永远长到「最高那页」那么大，内容少的页照样留一大片空白。
         *   取「内容高度」和「面板上限」里较小的值设成 height：
         *     · 内容放得下 → 面板就是内容那么高
         *     · 内容放不下 → 面板停在 cap，多出的内容进滑条
         *   ⚠ 用 height 不用 min-height：height 是定死，min-height 是至少，

         *   ⚠ cap 取的是 #mob-sheet 的 max-height，已在 place() 里按系数算过。
         *   ⚠⚠ 「更高」模式下不许锁 .mob-panes 的高度：

         *     判断要读实时 classList，不能在别处缓存。
         */
        const cap = parseFloat(sheet.style.maxHeight) || 1e9;
        if (wrap) {
          wrap.style.minHeight = '';                       // 清掉旧规则留下的 min-height
          if (sheet.classList.contains('mob-tall')) {
            /* 「更高」：不锁，交给 .mob-panes 用 flex 撑满剩余空间 */
            wrap.style.height = '';
          } else {
            /* 普通：取「内容最高页」与「面板上限」中较小者，多出的进滑条 */
            wrap.style.height = Math.round(Math.min(gDisp._lockHMax, cap)) + 'px';
          }
        }
      } catch (e) { /* 忽略 */ } finally {
        if (!wasMeasuring) { try { sheet.classList.remove('measuring'); } catch (e) { /* 忽略 */ } }
      }
    };
    // 开面板时量两次：立刻一次（多数页面此时已定），150ms 后再补一次

         * （class 原本只在 setSheet 打开面板时套，点了要等下次打开才生效）
         * 只有「正开着这一页」才需要动它，免得串到别的页。
         * ⚠ 这里的 opened === key 判断必须保留：放开成无条件会连带影响设置页的按钮。 */
        try { if (opened === key) sheet.classList.toggle('mob-tall', !!TALLS[key]); } catch (e) { /* 忽略 */ }

        try { if (gDisp._lockH) gDisp._lockH(); } catch (e) { /* 忽略 */ }
      });
      try { TALLS[key] = localStorage.getItem(tallStorageKey(key)) === '1'; } catch (e) { TALLS[key] = false; }
      paint();
      return b;

  wait(() => {

    th.classList.remove('mob-btn', 'mob-theme');
    if (!th.classList.contains('theme-btn')) th.classList.add('theme-btn');

    /* 已经在面板顶部就什么都不做（幂等）。 */
    let bar = gDisp.querySelector('.mob-theme-bar');
    if (bar && bar.contains(th)) return;

    /* 造那一行（没有才造） */
    if (!bar) {
      bar = document.createElement('div');
      bar.className = 'mob-theme-bar';
      const lab = document.createElement('span');
      lab.className = 'mob-theme-lab';

      else gDisp.insertBefore(bar, gDisp.firstChild);
    }
    bar.appendChild(th);
  });

  /* ============================================================ 手势 */
  const dist = (a, b) => Math.hypot(a.clientX - b.clientX, a.clientY - b.clientY);
  let two = null;
  stage.addEventListener('touchstart', (e) => {
    if (_exportDlg.picking) return;                          // 导出框选：双指也不缩放
    if (e.touches.length === 2) {
      two = {
        d: dist(e.touches[0], e.touches[1]),
        mx: (e.touches[0].clientX + e.touches[1].clientX) / 2,
        my: (e.touches[0].clientY + e.touches[1].clientY) / 2,
        /* 手势起始快照：记下两指刚落下时的 cam 和 scale，之后每帧从它重算，不做增量累积。
         * 只写 cx/cy 而不一起管 scale 时，缩放是合成 wheel 事件交给上层做的，
         * 那一层自己也会改 cam，两个写入者互相覆盖，画面就跳。
         * 这里只有一个写入者：zoomBy（项目里现成的锚点缩放）。 */
        cx: (state.cam && state.cam.cx) || 0,
        cy: (state.cam && state.cam.cy) || 0,
        sc: (state.cam && state.cam.scale) || 1,
      };
      e.preventDefault();
    }
  }, { passive: false });
  stage.addEventListener('touchmove', (e) => {
    if (_exportDlg.picking) return;                          // 导出框选：双指也不缩放
    if (e.touches.length !== 2 || !two) return;
    const d = dist(e.touches[0], e.touches[1]);
    const mx = (e.touches[0].clientX + e.touches[1].clientX) / 2;
    const my = (e.touches[0].clientY + e.touches[1].clientY) / 2;
    /* 双指缩放采用增量写法，不改成「从起始重算」。
     *   缩放本身是合成 wheel 事件交给上层去做，那一层也会改 cam，
     *   两边都动 cam 时谁覆盖谁看运气，表现就是缩放异常。
     *   所以缩放这块保持增量，想让它更稳只能从别处下手。
     *
     * 画面飘的两个原因：
     *  ① 增量累积：每帧 cam.cx -= dx / 当前 scale，而缩放一直在改 scale，
     *     补偿量每帧都变，误差滚雪球。
     *  ② 两个写入者：合成 wheel 事件交给上层，那层也写 cam，
     *     本层的平移与它的锚点补偿互相覆盖。
     * 处理方式：平移 = 起始 cam 减去（中点位移 / 起始 scale），是绝对量；
     *       缩放 = 先把 scale 还原成起始值，再调 zoomBy(总比, 起始中点)。
     *       zoomBy 内部先量 before、改 scale、再补 after，锚点保持不动。
     * 阈值取 0.01：两指距离变化千分之二属于手指微抖，不应触发缩放。
     * 不要改回合成 wheel 事件，也不要改回逐帧增量。
     */
    const ratio = d / (two.d || d);
    state.cam.cx = two.cx - (mx - two.mx) / two.sc;
    state.cam.cy = two.cy - (my - two.my) / two.sc;
    state.cam.scale = two.sc;
    if (Math.abs(ratio - 1) > 0.01 && typeof zoomBy === 'function') {
      zoomBy(ratio, two.mx, two.my);
    }
    /* 不回写 two.d/mx/my：它们是起始值，回写就退化成增量 */
    invalidate();
    e.preventDefault();
  }, { passive: false });
  const endTwo = () => { two = null; };
  stage.addEventListener('touchend', endTwo);
  stage.addEventListener('touchcancel', endTwo);

  /* ---- 单指涂色：自己驱动，不等浏览器合成的鼠标事件 ------------------
   * 手机上按住拖动时，浏览器不会持续派发 mousemove，
   * 桌面那套「按住拖一片」靠的就是 mousemove，所以在手机上只涂到第一块就断。
   * 这里自己接管：滑动就跟着 actAt，用的是同一个函数，逻辑不重写。
   *
   * 落笔时机：
   *   拖动 → 滑动时落笔（一笔到底）
   *   轻点 → 抬手时落那一块
   *   长按 → 只取色，不涂掉一块
   */
  let stroke = false, strokeAt = null, strokePainted = false, holdPicked = false;
  const stopStroke = () => { stroke = false; strokePainted = false; painting = false; };
  // 点在按钮 / 启动页 / 弹窗上时不抢事件，那些地方有自己的点击
  /* 这份清单与 app.js 的 onWidget 是同一份，加东西时两边一起加。
   *   含 #flag-pop / #flag-knobbox：旗帜浮窗也是控件，
   *   不挡的话，拖浮窗会在地图上涂一笔，点到浮窗背后的地块。 */
  const onWidget = (t) => !!(t && t.closest
    && t.closest('.zoom-box, button, input, a, select, #boot, .modal, #mob-ui, .toast,'
      + ' #flag-pop, #flag-knobbox'));

  /* 查看模式下，单指拖动 = 平移地图。
   * 桌面版靠 mousemove 平移，手机上不存在该事件，
   * 「查看」工具在手机上单指原本什么都做不了。
   * 这里自己接上，与双指平移用同一套算法（cam.cx -= dx / scale）。
   */
  let viewPan = null;
  let fillPending = null;      // 单击式工具（填色）按下的位置：一动就转平移

  stage.addEventListener('touchstart', (e) => {
    if (e.touches.length !== 1) { stopStroke(); viewPan = null; return; }  // 两根手指是平移缩放
    const t = e.touches[0];
    if (onWidget(e.target)) return;
    if (_exportDlg.picking) return;                          // 导出框选：画布工具全停
    if (!state.tool) return;
    if (state.tool === 'view') {                             // 查看：单指平移
      viewPan = { x: t.clientX, y: t.clientY };
      e.preventDefault();
      return;
    }
    stroke = true; strokePainted = false; holdPicked = false;
    painting = false;                                        // 还没落笔
    strokeAt = { x: t.clientX, y: t.clientY };
    /* 单击式工具（填色）：拖动就当作移动视线，与「查看」一致。
     *   填色的语义是「点一下动一次」，按住拖没有连续涂的意义。
     *   所以先当点击记下来，手指移动超过阈值就转成平移并取消这次落笔。
     *   判据用手指移动距离（不是时间），慢一点的点按仍然算点按。 */
    fillPending = (state.tool === 'fill') ? { x: t.clientX, y: t.clientY } : null;
    e.preventDefault();                                      // 挡掉合成鼠标事件，免得涂两遍
  }, { passive: false });

  stage.addEventListener('touchmove', (e) => {
    if (_exportDlg.picking) return;                          // 导出框选：不涂不缩放
    if (e.touches.length !== 1) return;
    const t = e.touches[0];
    if (viewPan) {                                           // 查看：拖到哪跟到哪
      const sc = (state.cam && state.cam.scale) || 1;
      state.cam.cx -= (t.clientX - viewPan.x) / sc;
      state.cam.cy -= (t.clientY - viewPan.y) / sc;
      viewPan.x = t.clientX; viewPan.y = t.clientY;
      invalidate();
      e.preventDefault();
      return;
    }
    if (fillPending) {                                       // 单击式工具：动了就转平移
      if (Math.hypot(t.clientX - fillPending.x, t.clientY - fillPending.y) > 10) {
        viewPan = { x: t.clientX, y: t.clientY };
        fillPending = null;
        stroke = false;                                      // 取消这次落笔
        strokePainted = true;                                // 抬手时不再补涂那一块
      }
      e.preventDefault();
      return;
    }
    if (!stroke) return;
    strokeAt = { x: t.clientX, y: t.clientY };
    if (state.tool !== 'pick') {                             // 吸管滑动不重复吸（与桌面一致）
      actAt(t.clientX, t.clientY);
      strokePainted = true;
      painting = true;
    }
    e.preventDefault();                                      // 手指在地图上时不让页面滚动
  }, { passive: false });

  const endStroke = () => {
    // 轻点一下（未滑动过）：抬手时把那一块涂上
    if (stroke && !strokePainted && !holdPicked && strokeAt) {
      try { actAt(strokeAt.x, strokeAt.y); } catch (err) { /* 忽略 */ }
    }
    strokeAt = null;
    viewPan = null;
    fillPending = null;
    stopStroke();
  };
  stage.addEventListener('touchend', endStroke);
  stage.addEventListener('touchcancel', () => { viewPan = null; fillPending = null; stopStroke(); });

  // 长按 = 右键 = 取色
  let lpTimer = null, lpTouched = null;
  const cancelLp = () => { if (lpTimer) { clearTimeout(lpTimer); lpTimer = null; } };
  stage.addEventListener('touchstart', (e) => {
    if (e.touches.length !== 1) { cancelLp(); return; }
    /* 旗帜 / 查看模式下长按不取色。
     *   旗帜的点击走 app.js 里的触摸分支，而长按会顺带改掉当前颜色并震动一下，
     *   只是点慢了一点就会触发；查看模式同理，查看只该是按住地图。 */
    if (state.tool === 'flag' || state.tool === 'view') { cancelLp(); return; }
    const t = e.touches[0];
    lpTouched = { x: t.clientX, y: t.clientY };
    cancelLp();
    lpTimer = setTimeout(() => {
      lpTimer = null;
      // 长按只取色，不涂掉一块；抬手时也不再补涂
      holdPicked = true;
      stroke = false;
      painting = false;
      try {
        stage.dispatchEvent(new MouseEvent('contextmenu', {
          clientX: lpTouched.x, clientY: lpTouched.y, button: 2, bubbles: true, cancelable: true,
        }));
        if (navigator.vibrate) navigator.vibrate(18);
      } catch (err) { /* 忽略 */ }
    }, 600);
  }, { passive: true });
  stage.addEventListener('touchmove', (e) => {
    if (!lpTouched || e.touches.length !== 1) return cancelLp();
    const t = e.touches[0];
    if (Math.abs(t.clientX - lpTouched.x) > 12 || Math.abs(t.clientY - lpTouched.y) > 12) cancelLp();
  }, { passive: true });
  stage.addEventListener('touchend', cancelLp);

  /* ============================================================ 进地图时：让高度充满屏幕
   * 手机是竖屏。若按「整张图塞进去」适配，2:1 的宽地图会被压成细细一条，
   * 所以手机上改为高度充满，宽度超出交给手指左右滑。
   * 电脑版保持原样：整张图一屏看全。
   */
  wait(() => {
    const _fit = fitView;
    if (typeof _fit !== 'function') return;
    fitView = function () {
      try {
        const v = viewRect();
        const m = state.meta;
        if (!v || !m || !m.mapHeight) return _fit();
        const s = (v.h * 0.985) / m.mapHeight;      // 高度刚好充满
        state.cam.scale = s;
        state.cam.cx = m.mapWidth / 2;
        state.cam.cy = m.mapHeight / 2;
        invalidate();
      } catch (e) { _fit(); }
    };
  });

  /* ============================================================ 当前工具 / 当前颜色（随时可见）
   * 手机上不打开面板就不知道当前是哪个工具、拿的什么颜色。
   * 内容每 0.4 秒与界面同步一次，不侵入 app 的逻辑，只读它的界面状态。
   */
  wait(() => {
    // 不再挂小胶囊：工具名已写在底部「工具」入口上，再挂一个重复
    const now = null;
    const sync = () => {
      try {
        const sw = $('brush-swatch');
        const dot = $('mob-now-sw');
        const bg = (sw && sw.style.background) || '#c83232';
        if (dot) dot.style.background = bg;
        const on = document.querySelector('[data-role="tool"] button.on');
        // 只取文字节点，不含隐藏的快捷键 <span class="k">E</span>；
        // 用 textContent 会把藏起来的字母也读出来，显示成「涂色E」
        let name = '工具';
        if (on) {
          let t = '';
          for (const n of on.childNodes) if (n.nodeType === 3) t += n.textContent;
          name = t.replace(/\s+/g, '') || String(on.textContent || '').replace(/[A-Za-z0-9]/g, '').trim() || '工具';
        }
        // 底部入口按钮直接写当前工具名，替换「工具」两个字
        const tb = $('mob-dock-tool');
        /* 图标固定用一支笔 ✎。曾试过每个工具换一个图标（emoji 彩色，与界面不协调），
         * 也试过自绘单色 SVG（观感不佳），最终回到最初的方案：
         * 图标固定，当前工具名写在旁边，靠文字区分。 */
        if (tb) tb.innerHTML = '<i>\u270e</i>' + name;
        // 底部渐变 = 当前颜色，涂色时能看到底色变化
        const dock = $('mob-dock');
        if (dock) {
          const c = getComputedStyle(sw || document.body).backgroundColor || '';
          const m2 = /rgba?\((\d+),\s*(\d+),\s*(\d+)/.exec(c);
          const rgb = m2 ? m2[1] + ',' + m2[2] + ',' + m2[3] : '200,50,50';
          dock.style.background = 'linear-gradient(180deg, rgba(' + rgb + ',0) 0%, rgba(' + rgb + ',.45) 42%, rgba(' + rgb + ',.82) 100%)';
        }
      } catch (e) { /* 忽略 */ }
    };
    sync();
    setInterval(sync, 400);
  });

  /* ============================================================ 撤销 / 重做 → 顶栏正中间
   * 这两项最常用，埋进「文件」面板里太深。

    const mid = $('mob-mid');

  /* ============================================================ 首次提示
   * 不再飘滚动提示条（「单指涂色 · 双指缩放平移 · 长按取色…」）：
   * 第一次进来会弹一页图文介绍，该说的都在里面；两条一起上反而打扰，
   * 手机上屏幕小，浮条还会挡地图。

  /* ============================================================ 去掉地图上的高亮

    renderer.setHover(null);
    /* 整片编组高亮是漏掉的那一路：
     * setHoverState(pid, paintOn, rgb, ?, ?, group) 共 6 个参数，
     * 最后一个是编组号（由 hoverGroupIdOf(pid) 算出，见 gl.js 的 rebuildHoverGroups）。
     * 上面三句都碰不到它：
     *   · renderer.hoverPid = 0      → 只管单块
     *   · renderer.hoverPaintOn = 0  → 只管同色整族
     *   · renderer.setHover(null)    → 也只管前五个
     * 这里再推一次全零，六个参数一个不留。 */
    if (renderer.setHoverState) renderer.setHoverState(0, 0, null, 0, null, 0);
  };
}

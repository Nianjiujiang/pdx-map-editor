
/* ================================================================ 手机版界面
 *
 * **只打进手机版** ✓ 电脑版源码一个字节不动 ✓
 *
 * 一级：地图 + 顶栏（左 返回 · 正中 永夜廿九 · 右 教程 / 导入导出）
 *             + 底部四个入口（显示设置 · 颜色 · 工具 · 视图）
 * 二级：**从你按的那个按钮正上方弹出来的那一栏** ✓（上拉菜单那种 ✓ 带个小尖角指着按钮）
 *       顶栏的「导入导出」因为按钮在上面，那一栏就往下弹 ✓
 *       「设置」的内容也在这个二级栏里 ✓ 不再套第三层 ✗
 *
 * 面板里的东西是**直接搬桌面现成的节点** ✓ 功能/状态/事件全沿用 ✓
 * 搬的过程每步单独兜错 ✓ 绝不连累按钮（踩过这个坑 ✗）
 */
function bindMobile() {
  const stage = $('stage');
  const app = $('app');
  if (!stage || !app) return;

  /* ============================================================ 一级界面 */
  const ui = document.createElement('div');
  ui.id = 'mob-ui';
  ui.innerHTML =
    '<div id="mob-backdrop"></div>'
    + '<div id="mob-topbar">'
    +   '<button type="button" class="mob-btn mob-back-arrow" id="mob-back" aria-label="回主菜单">\u2190</button>'
    +   '<span id="mob-mid"></span>'
    +   '<button type="button" class="mob-btn" id="mob-io">文件</button>'
    + '</div>'
    + '<div id="mob-dock">'
    +   '<button type="button" data-sheet="display"><i>\u2630</i>设置</button>'
    +   '<button type="button" data-sheet="color"><i>\u25c9</i>颜色</button>'
    +   '<button type="button" data-sheet="tool" id="mob-dock-tool"><i>\u270e</i>工具</button>'
    +   '<button type="button" data-sheet="view"><i>\u25a6</i>视图</button>'
    + '</div>'
    + '<div id="mob-sheet">'
    +   '<div id="mob-sheet-head"><b id="mob-sheet-title">面板</b><span></span>'
    +     '<button type="button" id="mob-sheet-x">收起</button></div>'
    +   '<div id="mob-sheet-body"></div>'
    + '</div>';
  app.appendChild(ui);

  const sheet = $('mob-sheet');
  const body = $('mob-sheet-body');
  const backdrop = $('mob-backdrop');
  const titleEl = $('mob-sheet-title');
  const wait = (fn) => { try { fn(); } catch (e) { if (window.console) console.warn('[手机版]', e && e.message); } };

  /* ============================================================ 面板骨架（先造好）
   * **顺序很重要** ✓：先把骨架和按钮的监听接上，再去搬节点 ✓
   * 这样就算某一步搬失败，点入口也照样出面板 ✓
   */
  const group = {};
  const makeGroup = (key, title) => {
    const g = document.createElement('div');
    g.className = 'mob-group';
    g.setAttribute('data-group', key);
    g.hidden = true;
    body.appendChild(g);
    group[key] = { el: g, title: title };
    return g;
  };
  const gColor = makeGroup('color', '颜色');
  const gTool = makeGroup('tool', '工具');
  const gView = makeGroup('view', '视图');
  const gDisp = makeGroup('display', '设置');
  const gIO = makeGroup('io', '文件');

  /** 把那一栏摆到按钮的正上方（按钮在上半屏就往下弹 ✓） */
  const place = (btn, key) => {
    try {
      const a = app.getBoundingClientRect();
      const r = btn.getBoundingClientRect();
      /* **每一栏按自己的内容定宽** ✓（用户定的）
       *   · 工具 / 视图 / 文件：里面就几个字或一排短按钮 → **窄**（别占满整屏 ✗）
       *   · 设置 / 颜色：一排排的开关和输入框 → **摊到接近整屏**（`a.width - 16` ✓）
       */
      const WANT = { tool: 148, view: 148, io: 148 };
      const FULL = { display: 1, color: 1 };
      const w = FULL[key] ? (a.width - 16) : Math.min(a.width - 16, WANT[key] || 300);
      const cx = r.left - a.left + r.width / 2;
      let left = cx - w / 2;
      left = Math.max(8, Math.min(a.width - w - 8, left));
      const below = (r.top - a.top) < a.height / 2;
      sheet.style.width = Math.round(w) + 'px';
      sheet.style.left = Math.round(left) + 'px';
      sheet.classList.toggle('below', below);
      if (below) {
        sheet.style.top = Math.round(r.bottom - a.top + 10) + 'px';
        sheet.style.bottom = 'auto';
        sheet.style.maxHeight = Math.max(160, a.height - (r.bottom - a.top) - 20) + 'px';
      } else {
        sheet.style.bottom = Math.round(a.bottom - r.top + 10) + 'px';
        sheet.style.top = 'auto';
        sheet.style.maxHeight = Math.max(160, r.top - a.top - 20) + 'px';
      }
      sheet.style.setProperty('--tail-x', Math.round(cx - left) + 'px');
      /* 「更高」模式：把 height 顶到**刚算出来的可用上限**（就是上面那个 max-height ✓
       *   贴着底部往上长 ✓ 不会顶出屏幕 ✓）；普通模式清掉 height ✓ 回到"按内容自己撑" ✓
       * 这里**不做动画** ✗（用户要求去掉 ✓ 而且显式 px 也会让面板不再跟内容长高 ✗）*/
      try {
        if (sheet.classList.contains('mob-tall') && sheet.style.maxHeight) {
          sheet.style.height = sheet.style.maxHeight;
        } else {
          sheet.style.height = '';
        }
      } catch (e) { /* ✓ */ }
    } catch (e) { /* 算不出来就按默认位置 ✓ */ }
  };

  let opened = null, openedBtn = null;
  const setSheet = (key, btn) => {
    const g = key ? group[key] : null;
    for (const k in group) group[k].el.hidden = !(g && group[k] === g);
    if (g) {
      titleEl.textContent = g.title;
      /* 「更高」**按页各记各的** ✓ —— 打开哪一页就套哪一页的状态 ✓
       * ⚠ 这一句**必须在 place() 之前** ✗：
       *   ① 少了它，按钮加上的 mob-tall 就没人摘 ✗ → 之后打开工具/视图/文件也全变高 ✗
       *      （整个 App 只有一个 #mob-sheet ✓ 四个入口共用 ✓ 用户报过两次 ✓）
       *   ② place() 是**看着这个 class** 决定高度的 ✓ 顺序反了就会按上一页的状态算一次 ✗
       */
      try {
        const _ts = (group.display && group.display.el && group.display.el._tallState) || {};
        sheet.classList.toggle('mob-tall', !!_ts[key]);
      } catch (e) { /* ✓ */ }
      if (btn) place(btn, key);
      // 高度**不锁** ✓ 跟「颜色」那栏一样按内容自己撑 ✓（用户要求）—— 这一句是给别的面板的
      //（设置那一栏例外：下面用 _lockSoon 量出最高页锁住 ✓ 切页不跳 ✓）
      sheet.classList.remove('tall');
      if (key === 'display' && group.display && group.display.el._lockSoon) {
        requestAnimationFrame(() => { try { group.display.el._lockSoon(); } catch (e) { /* ✓ */ } });
      }
      // **横向滚动位置强制归零** ✓（不然内容稍宽就被推到一边、左边那截看不见 ✗）
      try { body.scrollLeft = 0; sheet.scrollLeft = 0; } catch (e) { /* ✓ */ }
      /* 打开设置面板时，要是当前正停在「导出图例」那一页，把列表重刷一遍 ✓
       * 用 legendFresh ✓ —— 它会**先按"现在这一刻的剧本/层级"把色块补算一遍**再列 ✓
       * （只刷列表的话，列出来的可能还是上一个剧本的分组 ✗ 用户报过 ✓）*/
      if (key === 'display') {
        const _dEl = group.display && group.display.el ? group.display.el : null;
        const _lgPane = _dEl ? _dEl.querySelector('.mob-pane[data-pane="legend"]') : null;
        if (_lgPane && !_lgPane.hidden) {
          try { legendFresh(); } catch (e) { /* ✓ */ }
        }
      }
      sheet.classList.add('on');
      backdrop.classList.add('on');
      opened = key;
      openedBtn = btn || null;
    } else {
      sheet.classList.remove('on');
      backdrop.classList.remove('on');
      opened = null;
      openedBtn = null;
    }
    const bs = ui.querySelectorAll('#mob-dock button');
    for (let i = 0; i < bs.length; i++) bs[i].classList.toggle('on', bs[i].getAttribute('data-sheet') === key);
    const io = $('mob-io');
    if (io) io.classList.toggle('on', key === 'io');
  };

  const dockWs = ui.querySelectorAll('#mob-dock button');
  for (let i = 0; i < dockWs.length; i++) {
    (function (b) {
      b.addEventListener('click', () => {
        const k = b.getAttribute('data-sheet');
        setSheet(opened === k ? null : k, b);
      });
    })(dockWs[i]);
  }
  const ioBtn = $('mob-io');
  if (ioBtn) ioBtn.addEventListener('click', () => setSheet(opened === 'io' ? null : 'io', ioBtn));
  const backBtn = $('mob-back');
  if (backBtn) backBtn.addEventListener('click', () => {
    // **只管一件事：回主菜单** ✓（用户定的 —— 不让它兼职关弹窗、收面板 ✗）
    // **而且不许把画的东西弄丢** ✓：先写进暂存区 ✓ 回来选同一张图会自动接上 ✓
    // （暂存是按地图分开存的 ✓ 换地图不会串味 ✓）
    let dirty = 0;
    try { dirty = (state.changed && state.changed.size) || 0; } catch (e) { dirty = 0; }
    if (dirty > 0 && !stashNow()) {
      // 存不进暂存区（无痕模式 / 配额满）才问一句 ✗ 至少不静默丢掉
      const ok = window.confirm('这台浏览器不让暂存，回主菜单会丢掉还没导出的涂色（改动 ' + dirty + ' 处）。\n'
        + '想留着就先点「文件 → 导出涂色」存一份。\n\n确定要回主菜单吗？');
      if (!ok) return;
    }
    // 换地图要把数据、渲染器整套重来 ✓ 页面自带的启动流程正好干这事 ✓
    // 重新加载之后，同一张图的涂色会由 loadProject() 自动接上 ✓
    location.reload();
  });
  if ($('mob-sheet-x')) $('mob-sheet-x').addEventListener('click', () => setSheet(null));
  if (backdrop) backdrop.addEventListener('click', () => setSheet(null));
  document.addEventListener('keydown', (e) => { if (e.key === 'Escape') setSheet(null); });
  // 横竖屏切换 / 转屏时，那一栏跟着按钮重新摆一下 ✓
  window.addEventListener('resize', () => { if (opened && openedBtn) place(openedBtn, opened); });

  /* ============================================================ 再搬节点 */
  const panelOf = (sel) => {
    const n = typeof sel === 'string' ? $(sel) : sel;
    return n && n.closest ? n.closest('section.panel') : null;
  };
  const move = (g, node) => { if (node && node.parentNode !== g) g.appendChild(node); };
  const movePanelOf = (g, sel) => { move(g, panelOf(sel)); };

  wait(() => {
    movePanelOf(gColor, 'brush-hex');        // 当前颜色：色块 / HEX / RGB / 标记
    movePanelOf(gColor, 'palette');          // 最近使用
    movePanelOf(gColor, 'results');          // 搜索
    move(gColor, $('label-pick'));
  });
  wait(() => {
    move(gTool, $('tool-group'));
    /* **工具选好就自动收起面板** ✓（用户要求：点完马上回到地图上动手 ✓）
     * 监听挂在工具按钮那一层 —— app 自己的 onclick 会**先把工具切好** ✓，
     * 冒泡到这儿我们再收面板 ✓（顺序天然对，不用去猜它绑在哪 ✓）
     */
    const tp = document.querySelector('[data-role="tool"]');
    if (tp) tp.addEventListener('click', (e) => {
      const b = e.target && e.target.closest ? e.target.closest('button') : null;
      if (b) setSheet(null);
    });
  });
  wait(() => { move(gView, $('tier-group')); });
  wait(() => {
    /* **「设置」= 一排标签按钮 + 下面跟着换内容** ✓（用户要的 ✓）
     *   标签：显示 / 颜色 / 边界 / 名字 / 其他
     *   内容：桌面设置里那四栏 + 显示开关
     *   （面板顶上那个多余的总标题行去掉了 ✗ 用户嫌重复 ✓）
     */
    const tabs = document.createElement('div');
    tabs.className = 'mob-tabs';
    gDisp.appendChild(tabs);

    const panes = document.createElement('div');
    panes.className = 'mob-panes';
    gDisp.appendChild(panes);

    const TABS = [
      { key: 'show', title: '显示' },
      { key: 'color', title: '颜色' },
      { key: 'border', title: '边界' },
      /* **名字自己一页** ✓（用户要求：别塞进「显示」✗）
       *   桌面那边它也已经是独立一栏了 ✓ 手机这儿跟着分开 ✓ */
      { key: 'name', title: '名字' },
      /* ⚠ key 仍叫 legend ✗（别改，页里还有几处按它找 DOM ✓）
       *   只把**标题**改成「其他」—— 跟桌面那栏对齐 ✓ */
      { key: 'legend', title: '其他' },
    ];
    const paneEls = {};
    let firstTab = null;
    for (const t of TABS) {
      const pane = document.createElement('div');
      pane.className = 'mob-pane';
      pane.setAttribute('data-pane', t.key);
      pane.hidden = true;
      panes.appendChild(pane);
      paneEls[t.key] = pane;
      const b = document.createElement('button');
      b.type = 'button';
      b.className = 'mob-tab';
      b.setAttribute('data-pane', t.key);
      b.textContent = t.title;
      tabs.appendChild(b);
      if (!firstTab) firstTab = b;
    }
    const showTab = (k) => {
      try { body.scrollLeft = 0; } catch (e) { /* ✓ */ }
      for (const key in paneEls) paneEls[key].hidden = (key !== k);
      const bs = tabs.querySelectorAll('.mob-tab');
      for (let i = 0; i < bs.length; i++) bs[i].classList.toggle('on', bs[i].getAttribute('data-pane') === k);
      /* **图例列表每次打开这一页都要重刷** ✓
       * 手机版的设置面板不是那个桌面弹窗 ✗（#settings 在手机上是 display:none，
       * 四栏是被**搬**进来的 ✓）→ 而"刷图例列表"原本是桌面弹窗打开时顺手做的 ✓
       * 于是手机版没人替它刷 → 那个列表**永远停在启动那一刻**（= 空的 ✗）
       * 用户报的"手机版设置里看不见图例"就是这个 ✓ */
      if (k === 'legend') { try { legendFresh(); } catch (e) { /* ✓ */ } }
      // 换了页，内容高度本来就变 → 不用管高度（普通模式是按内容自己撑的 ✓）
      /* **每次切页再量一次** ✓ —— 万一这一页比之前量过的都高（比如列表这次更长 ✓）
       * 就把锁的高度抬上去 ✓（只涨不缩 ✓ 所以不会再出现"切回去变矮"✗）*/
      try { if (gDisp._lockH) setTimeout(() => { try { gDisp._lockH(); } catch (e) { /* ✓ */ } }, 0); }
      catch (e) { /* ✓ */ }
    };
    for (const b of tabs.querySelectorAll('.mob-tab')) {
      b.addEventListener('click', () => showTab(b.getAttribute('data-pane')));
    }
    showTab('show');

    // **切换标签不许改高度** ✓（用户嫌一跳一跳的 ✗ 前后提过两次 ✗）
    // 做法：把每页都临时显出来（用 visibility:hidden 保住布局 ✓）量高度 ✓
    //       取**见过的最大高度**当内容区的 min-height 锁住 ✓ → 只涨不缩 ✓
    // 为什么要"只涨不缩"：光量一次不够 ✗ ——
    //   图例列表要现填、字体换行要重排，页面是**过一下才长高**的 ✓
    //   只量一次就会锁在"刚打开那会儿最矮的值"✗ → 之后切到高的页照样长出去 ✓
    // 所以：记住历史最大值 ✓ + 开面板时量两次（立刻 + 150ms）✓ + 每次切页再量一次 ✓
    gDisp._lockHMax = 0;
    gDisp._lockH = () => {
      const wasMeasuring = sheet.classList.contains('measuring');
      try {
        const panes = tabs.parentNode.querySelectorAll('.mob-pane');
        if (!panes.length) return;
        // 量之前先临时收回图例列表的上限 ✓
        //（不然在「更高」模式下量一次，锁就被顶到 42vh 那么高 ✗ 普通模式也跟着变高 ✗）
        sheet.classList.add('measuring');
        const act = [];
        for (const p2 of panes) { act.push(p2.hidden); p2.hidden = false; p2.style.visibility = 'hidden'; }
        let max = 0;
        for (const p2 of panes) max = Math.max(max, p2.offsetHeight || p2.scrollHeight || 0);
        for (let i = 0; i < panes.length; i++) { panes[i].style.visibility = ''; panes[i].hidden = act[i]; }
        if (max <= 0) return;
        gDisp._lockHMax = Math.max(gDisp._lockHMax || 0, max);   // **只涨不缩** ✓
        const wrap = gDisp.querySelector('.mob-panes');
        if (wrap) wrap.style.minHeight = Math.round(gDisp._lockHMax) + 'px';
      } catch (e) { /* ✓ */ } finally {
        if (!wasMeasuring) { try { sheet.classList.remove('measuring'); } catch (e) { /* ✓ */ } }
      }
    };
    // 开面板时量两次：立刻一次（大部分页面这时就定了 ✓）、150ms 后再补一次
    //（补的是"现填内容/重排之后"的高度 ✓ 比如图例列表刚列出来那一刻 ✓）
    gDisp._lockSoon = () => {
      try { gDisp._lockH(); } catch (e) { /* ✓ */ }
      setTimeout(() => { try { gDisp._lockH(); } catch (e) { /* ✓ */ } }, 150);
    };

    /* **「更高」按钮：按页各记各的** ✓（设置页一个、颜色页一个 ✓）
     * 为什么改成"按页各记各的"：整个 App 只有一个 #mob-sheet ✗ 四个入口共用 ✓
     *   状态混在一起的话，在设置页点了变高 → 切到颜色页**也跟着变高** ✗（用户报过 ✓）
     * 现在每页自己的开关记在自己的键里 ✓ 打开哪页就套哪页的状态 ✓ 天然不串 ✓
     * 高度**不写死** ✓ JS 会顶到 place() 算出来的可用上限（顶到极限但不顶出屏幕 ✓）
     */
    const TALLS = {};                       // { display: true, color: false, ... }
    const tallStorageKey = (k) => 'pdx_mob_tall_' + k;
    const makeTallBtn = (host, key) => {
      if (!host) return null;
      const b = document.createElement('button');
      b.type = 'button';
      b.className = 'mob-tallbtn';
      b.setAttribute('aria-label', '把面板变高或变矮');
      host.appendChild(b);
      const paint = () => {
        const on = !!TALLS[key];
        b.classList.toggle('on', on);
        // ▾ = 还能再高一点；▴ = 已经最高了，点了收回去 ✓（用户要求用上下箭头 ✓）
        b.textContent = on ? '\u25b4' : '\u25be';
      };
      b.addEventListener('click', () => {
        TALLS[key] = !TALLS[key];
        try { localStorage.setItem(tallStorageKey(key), TALLS[key] ? '1' : '0'); } catch (e) { /* ✓ */ }
        paint();
        /* **当场把 class 套上去** ✗ —— 漏了这一步按钮就跟没反应一样 ✓
         * （class 原本只在 setSheet 打开面板时套 ✓ 所以点了要等下次打开才生效 ✗）
         * 只有"正开着这一页"才需要动它 ✓ 免得串到别的页 ✓ */
        try { if (opened === key) sheet.classList.toggle('mob-tall', !!TALLS[key]); } catch (e) { /* ✓ */ }
        // 当前正开着这一页才需要立刻重摆 ✓（place 里会把高度一起收拾好 ✓）
        try { if (opened === key && openedBtn) place(openedBtn, key); } catch (e) { /* ✓ */ }
        try { if (key === 'display' && gDisp._lockH) gDisp._lockH(); } catch (e) { /* ✓ */ }
      });
      try { TALLS[key] = localStorage.getItem(tallStorageKey(key)) === '1'; } catch (e) { TALLS[key] = false; }
      paint();
      return b;
    };
    gDisp._tallState = TALLS;               // 给 setSheet 用 ✓
    makeTallBtn(tabs, 'display');           // 设置页：按钮放标签那一排的最右边 ✓

    // 颜色页：按钮放进它自己的「当前颜色」标题行里 ✓（那一行本来就是一条 header ✓）
    try {
      const head = gColor.querySelector('.panel-head');
      if (head) {
        head.style.display = 'flex';
        head.style.alignItems = 'center';
        head.style.gap = '8px';
        makeTallBtn(head, 'color');
      } else {
        makeTallBtn(gColor, 'color');
      }
    } catch (e) { /* ✓ */ }

    /* **没有荒地的地图：把「荒地」整组藏掉** ✓
     * 判断依据跟教程模块用的是同一个：app 把 #waste-label 的内联 display 设成 none ✓
     *（这一组在手机上会被那条 group-label 规则顶出一个孤零零的标题 ✗ 见 CSS 里的说明 ✓）
     */
    try {
      const wl2 = $('waste-label');
      const wg2 = $('waste-group');
      const hiddenByApp = !!(wl2 && wl2.style && wl2.style.display === 'none');
      if (wg2 && hiddenByApp) wg2.classList.add('no-waste');
    } catch (e) { /* ✓ */ }

    // 颜色页变高时让"搜索结果"那块吃掉空间 ✓ 先给它挂个记号（CSS 靠这个记号找它 ✓）
    try {

  /* ============================================================ 手势 */
  const dist = (a, b) => Math.hypot(a.clientX - b.clientX, a.clientY - b.clientY);
  let two = null;
  stage.addEventListener('touchstart', (e) => {
    if (e.touches.length === 2) {
      two = {
        d: dist(e.touches[0], e.touches[1]),
        mx: (e.touches[0].clientX + e.touches[1].clientX) / 2,
        my: (e.touches[0].clientY + e.touches[1].clientY) / 2,
      };
      e.preventDefault();
    }
  }, { passive: false });
  stage.addEventListener('touchmove', (e) => {
    if (e.touches.length !== 2 || !two) return;
    const d = dist(e.touches[0], e.touches[1]);
    const mx = (e.touches[0].clientX + e.touches[1].clientX) / 2;
    const my = (e.touches[0].clientY + e.touches[1].clientY) / 2;
    const sc = (state.cam && state.cam.scale) || 1;
    state.cam.cx -= (mx - two.mx) / sc;
    state.cam.cy -= (my - two.my) / sc;
    const ratio = d / (two.d || d);
    if (Math.abs(ratio - 1) > 0.002) {
      // 合成一个**真实的 wheel 事件** ✓ 现成的缩放逻辑和教程那一步都认它 ✓
      try {
        stage.dispatchEvent(new WheelEvent('wheel', {
          deltaY: -Math.log(ratio) / 0.0016, clientX: mx, clientY: my,
          bubbles: true, cancelable: true,
        }));
      } catch (err) { /* 老浏览器就只平移 ✓ */ }
    }
    two.d = d; two.mx = mx; two.my = my;
    invalidate();
    e.preventDefault();
  }, { passive: false });
  const endTwo = () => { two = null; };
  stage.addEventListener('touchend', endTwo);
  stage.addEventListener('touchcancel', endTwo);

  /* ---- 单指涂色：**自己驱动**，不等浏览器合成的鼠标事件 ------------------
   * 手机上按住拖动时，浏览器**不会持续派发 mousemove** ✗
   * 桌面那套"按住拖一片"靠的就是 mousemove → 所以在手机上只涂到第一块就断 ✓
   * 这里自己接管：滑动就跟着 actAt ✓ 用的是**同一个函数** ✓ 逻辑不重写
   *
   * 落笔时机也顺便改聪明了：
   *   拖动 → 滑动时落笔（一笔到底 ✓）
   *   轻点 → 抬手时落那一块 ✓
   *   长按 → 只取色 ✓ **不再顺手涂掉一块**（以前的老毛病 ✓ 现在治好了）
   */
  let stroke = false, strokeAt = null, strokePainted = false, holdPicked = false;
  const stopStroke = () => { stroke = false; strokePainted = false; painting = false; };
  // 点在按钮 / 启动页 / 弹窗上时不要抢 ✓（那些地方有它们自己的点击）
  const onWidget = (t) => !!(t && t.closest
    && t.closest('.zoom-box, button, input, a, select, #boot, .modal, #mob-ui, .toast'));

  /* **查看模式下，单指拖动 = 平移地图** ✓
   * 桌面版是靠 mousemove 平移的 ✗ 手机上根本没那东西 ✓
   * 所以「查看」工具在手机上原本单指什么都干不了 ✗（用户报的第三条）
   * 这里自己接上：跟双指平移用**同一套算法**（cam.cx -= dx / scale ✓）
   */
  let viewPan = null;

  stage.addEventListener('touchstart', (e) => {
    if (e.touches.length !== 1) { stopStroke(); viewPan = null; return; }  // 两根手指是平移缩放 ✓
    const t = e.touches[0];
    if (onWidget(e.target)) return;
    if (!state.tool) return;
    if (state.tool === 'view') {                             // 查看：单指平移 ✓
      viewPan = { x: t.clientX, y: t.clientY };
      e.preventDefault();
      return;
    }
    stroke = true; strokePainted = false; holdPicked = false;
    painting = false;                                        // 还没落笔 ✓
    strokeAt = { x: t.clientX, y: t.clientY };
    e.preventDefault();                                      // 挡掉合成鼠标事件 ✓ 免得涂两遍
  }, { passive: false });

  stage.addEventListener('touchmove', (e) => {
    if (e.touches.length !== 1) return;
    const t = e.touches[0];
    if (viewPan) {                                           // 查看：拖到哪跟到哪 ✓
      const sc = (state.cam && state.cam.scale) || 1;
      state.cam.cx -= (t.clientX - viewPan.x) / sc;
      state.cam.cy -= (t.clientY - viewPan.y) / sc;
      viewPan.x = t.clientX; viewPan.y = t.clientY;
      invalidate();
      e.preventDefault();
      return;
    }
    if (!stroke) return;
    strokeAt = { x: t.clientX, y: t.clientY };
    if (state.tool !== 'pick') {                             // 吸管滑动不重复吸 ✓（跟桌面一致）
      actAt(t.clientX, t.clientY);
      strokePainted = true;
      painting = true;
    }
    e.preventDefault();                                      // 手指在地图上就别让页面滚 ✓
  }, { passive: false });

  const endStroke = () => {
    // 轻点一下（没滑动过）→ 抬手时把那一块涂上 ✓
    if (stroke && !strokePainted && !holdPicked && strokeAt) {
      try { actAt(strokeAt.x, strokeAt.y); } catch (err) { /* ✓ */ }
    }
    strokeAt = null;
    viewPan = null;
    stopStroke();
  };
  stage.addEventListener('touchend', endStroke);
  stage.addEventListener('touchcancel', () => { viewPan = null; stopStroke(); });

  // 长按 = 右键 = 取色 ✓
  let lpTimer = null, lpTouched = null;
  const cancelLp = () => { if (lpTimer) { clearTimeout(lpTimer); lpTimer = null; } };
  stage.addEventListener('touchstart', (e) => {
    if (e.touches.length !== 1) { cancelLp(); return; }
    const t = e.touches[0];
    lpTouched = { x: t.clientX, y: t.clientY };
    cancelLp();
    lpTimer = setTimeout(() => {
      lpTimer = null;
      // **长按只取色，别顺手涂掉一块** ✓ 抬手时也不再补涂 ✓
      holdPicked = true;
      stroke = false;
      painting = false;
      try {
        stage.dispatchEvent(new MouseEvent('contextmenu', {
          clientX: lpTouched.x, clientY: lpTouched.y, button: 2, bubbles: true, cancelable: true,
        }));
        if (navigator.vibrate) navigator.vibrate(18);
      } catch (err) { /* ✓ */ }
    }, 600);
  }, { passive: true });
  stage.addEventListener('touchmove', (e) => {
    if (!lpTouched || e.touches.length !== 1) return cancelLp();
    const t = e.touches[0];
    if (Math.abs(t.clientX - lpTouched.x) > 12 || Math.abs(t.clientY - lpTouched.y) > 12) cancelLp();
  }, { passive: true });
  stage.addEventListener('touchend', cancelLp);

  /* ============================================================ 进地图时：让**高**充满屏幕
   * 手机是竖屏 ✓ 要是按"整张图塞进去"来适配，那张 2:1 的宽地图会被压成细细一条 ✗
   * 所以手机上改成**高度充满** ✓ 宽度超出就让手指左右滑 ✓ 看得清多了 ✓
   * （电脑版保持原样：整张图一屏看全 ✓）
   */
  wait(() => {
    const _fit = fitView;
    if (typeof _fit !== 'function') return;
    fitView = function () {
      try {
        const v = viewRect();
        const m = state.meta;
        if (!v || !m || !m.mapHeight) return _fit();
        const s = (v.h * 0.985) / m.mapHeight;      // 高度刚好充满 ✓
        state.cam.scale = s;
        state.cam.cx = m.mapWidth / 2;
        state.cam.cy = m.mapHeight / 2;
        invalidate();
      } catch (e) { _fit(); }
    };
  });

  /* ============================================================ 当前工具 / 当前颜色（随时可见）

   * 内容每 0.4 秒跟界面同步一次 ✓（不侵入 app 的逻辑 ✓ 只读它的界面状态 ✓）
   */
  wait(() => {
    // 这一版**不再挂那个小胶囊**了 ✓（工具名已经写在底部「工具」入口上 ✓ 再挂一个重复 ✗）
    const now = null;
    const sync = () => {
      try {
        const sw = $('brush-swatch');

        const bg = (sw && sw.style.background) || '#c83232';

        //（之前用 textContent 把藏起来的字母也读出来了 → 显示成"涂色E" ✗）

          name = t.replace(/\s+/g, '') || String(on.textContent || '').replace(/[A-Za-z0-9]/g, '').trim() || '工具';
        }
        // **底部那个入口按钮也直接写当前工具名** ✓（用户要求：把"工具"两个字换掉 ✓）
        const tb = $('mob-dock-tool');
        /* ⚠ **图标固定就是一支笔 ✎**（用户定的 ✓）

         *   当前工具名仍然写在旁边 ✓ 靠文字区分就够了 ✓ */
        if (tb) tb.innerHTML = '<i>\u270e</i>' + name;
        // **底部渐变 = 当前颜色** ✓（用户要求 ✓ 一边涂一边看着底色变 ✓）
        const dock = $('mob-dock');

          const m2 = /rgba?\((\d+),\s*(\d+),\s*(\d+)/.exec(c);
          const rgb = m2 ? m2[1] + ',' + m2[2] + ',' + m2[3] : '200,50,50';
          dock.style.background = 'linear-gradient(180deg, rgba(' + rgb + ',0) 0%, rgba(' + rgb + ',.45) 42%, rgba(' + rgb + ',.82) 100%)';

  /* ============================================================ 教程：手机版改成"一个介绍弹窗"

   */
  const introHTML =
    '<p class="mi-lead">把 P社五款游戏的<strong>真地图</strong>涂颜色、改国界、导出成图 —— 就在这个网页里，不用装游戏、不用注册、也不用联网。</p>'

// ================================================================ 手机端：点地块不高亮
// 触屏没有"把鼠标移开"这个动作 —— 手指点一下，高亮就黏在屏幕上消不掉 ✗
// 看着像"这块被选中了"，所以手机端**任何情况下都不点亮**（信息卡照常出 ✓）
{
  const _hlDesk = applyHoverHighlight;
  applyHoverHighlight = function (pid) {
    _hlDesk(pid || 0, null);     // 状态照旧更新：卡片 / 光标逻辑还要用 ✓
    renderer.hoverPid = 0;       // 单块高亮：关 ✓
    renderer.hoverPaintOn = 0;   // 整族高亮（同色一族一起亮）：也关 ✓
    renderer.setHover(null);
  };
}


/* ================================================================ 手机版界面
 *
 * **只打进手机版** ✓ 电脑版源码一个字节不动 ✓
 *
 * 一级：地图 + 顶栏（左 返回 · 正中 永夜廿九 · 右 教程 / 导入导出）
 *             + 底部四个入口（显示设置 · 颜色 · 工具 · 视图）
 * 二级：**从你按的那个按钮正上方弹出来的那一栏** ✓（上拉菜单那种 ✓ 带个小尖角指着按钮）
 *       顶栏的「导入导出」因为按钮在上面，那一栏就往下弹 ✓
 *       「设置」的内容也在这个二级栏里 ✓ 不再套第三层 ✗
 *
 * 面板里的东西是**直接搬桌面现成的节点** ✓ 功能/状态/事件全沿用 ✓
 * 搬的过程每步单独兜错 ✓ 绝不连累按钮（踩过这个坑 ✗）
 */
function bindMobile() {
  const stage = $('stage');
  const app = $('app');
  if (!stage || !app) return;

  /* ============================================================ 一级界面 */
  const ui = document.createElement('div');
  ui.id = 'mob-ui';
  ui.innerHTML =
    '<div id="mob-backdrop"></div>'
    + '<div id="mob-topbar">'
    +   '<button type="button" class="mob-btn mob-back-arrow" id="mob-back" aria-label="回主菜单">\u2190</button>'
    +   '<span id="mob-mid"></span>'
    +   '<button type="button" class="mob-btn" id="mob-io">文件</button>'
    + '</div>'
    + '<div id="mob-dock">'
    +   '<button type="button" data-sheet="display"><i>\u2630</i>设置</button>'
    +   '<button type="button" data-sheet="color"><i>\u25c9</i>颜色</button>'
    +   '<button type="button" data-sheet="tool" id="mob-dock-tool"><i>\u270e</i>工具</button>'
    +   '<button type="button" data-sheet="view"><i>\u25a6</i>视图</button>'
    + '</div>'
    + '<div id="mob-sheet">'
    +   '<div id="mob-sheet-head"><b id="mob-sheet-title">面板</b><span></span>'
    +     '<button type="button" id="mob-sheet-x">收起</button></div>'
    +   '<div id="mob-sheet-body"></div>'
    + '</div>';
  app.appendChild(ui);

  const sheet = $('mob-sheet');
  const body = $('mob-sheet-body');
  const backdrop = $('mob-backdrop');
  const titleEl = $('mob-sheet-title');
  const wait = (fn) => { try { fn(); } catch (e) { if (window.console) console.warn('[手机版]', e && e.message); } };

  /* ============================================================ 选文件：过滤整个放开
   * ⚠ 用户报的"手机上没法把 pdxmap 识别为可导入文件"（2026-10-12）✗：
   *   `.pdxmap` 是自定义扩展名，系统不认 → 报给选择器的类型是 `application/octet-stream` ✓
   *   而安卓 / iOS 的文件选择器**按 MIME 过滤** ✗ → 文件灰着，点都点不动 ✓
   *   （源头那份 accept 已经补了 octet-stream，这里再兜一道：手机上一律不过滤 ✓
   *    反正选错了也进不来 —— pack.js 的校验器会说清楚哪个文件、哪儿不对 ✓）
   */
  wait(() => {
    const pf = $('pack-file');
    if (pf && pf.accept !== '*/*') pf.accept = '*/*';
  });

  /* ============================================================ 面板骨架（先造好）
   * **顺序很重要** ✓：先把骨架和按钮的监听接上，再去搬节点 ✓
   * 这样就算某一步搬失败，点入口也照样出面板 ✓
   */
  const group = {};
  const makeGroup = (key, title) => {
    const g = document.createElement('div');
    g.className = 'mob-group';
    g.setAttribute('data-group', key);
    g.hidden = true;
    body.appendChild(g);
    group[key] = { el: g, title: title };
    return g;
  };
  const gColor = makeGroup('color', '颜色');
  const gTool = makeGroup('tool', '工具');
  const gView = makeGroup('view', '视图');
  const gDisp = makeGroup('display', '设置');
  const gIO = makeGroup('io', '文件');

  /** 把那一栏摆到按钮的正上方（按钮在上半屏就往下弹 ✓） */
  const place = (btn, key) => {
    try {
      const a = app.getBoundingClientRect();
      const r = btn.getBoundingClientRect();
      /* **每一栏按自己的内容定宽** ✓（用户定的）
       *   · 工具 / 视图 / 文件：里面就几个字或一排短按钮 → **窄**（别占满整屏 ✗）
       *   · 设置 / 颜色：一排排的开关和输入框 → **摊到接近整屏**（`a.width - 16` ✓）
       */
      const WANT = { tool: 148, view: 148, io: 148 };
      const FULL = { display: 1, color: 1 };
      const w = FULL[key] ? (a.width - 16) : Math.min(a.width - 16, WANT[key] || 300);
      const cx = r.left - a.left + r.width / 2;
      let left = cx - w / 2;
      left = Math.max(8, Math.min(a.width - w - 8, left));
      const below = (r.top - a.top) < a.height / 2;
      sheet.style.width = Math.round(w) + 'px';
      sheet.style.left = Math.round(left) + 'px';
      sheet.classList.toggle('below', below);
      /* 🔴 **两次三成**（用户 2026-10-13："设置高度减去 30%"→ 之后又说"再缩小 30% 左右"✓）
       *   最终系数 = 0.7 × 0.7 ≈ **0.5** ✓（取整好记，跟"再缩三成"差 1% ✓）
       *   ⚠ 这两句算出来的是**"从按钮一路到屏幕边缘"的可用高度** ✗
       *     而它写成**内联样式** ✓ 优先级压过 `mobile_css.txt` 里的
       *     `#mob-sheet { max-height: … }` ✓（所以光改 CSS 一点用都没有 ✗
       *       用户连着报了好几回"没看到"✓ 就是这个原因 ✓）
       *   ⚠ 必须**在这儿**乘 ✗ 不能在 CSS 里补 ✓
       *   剩下的空间给地图 ✓ 放不下的内容交给 `#mob-sheet-body` 的滑条 ✓
       *   ⚠ 改系数时**顺手把 css 那两处 vh 也改一下** ✗ 它们只是兜底（被内联压着 ✓）
       *     但两边数值差太远容易看糊涂 ✓ */
      const _avail = (below ? (a.height - (r.bottom - a.top) - 20) : (r.top - a.top - 20));
      /* 🔴 **「更高」模式：上限回到"可用高度"，不再乘系数** ✗✗（2026-10-13 修 ✓）
       *   这是"小按钮没用"的**真正原因** ✗ ——
       *   我把 cap 压到 0.5×可用之后，面板的 `height` 和 `max-height` **都是这个 cap** ✓
       *   于是按不按「更高」**面板都是同一个数** ✓ 按钮看着毫无反应 ✓
       *   （我前两轮还在折腾 `wrap.style.height` 和 flex ✗ 都是白费力气 ✓
       *     那个按钮的语义本来就是"**面板整体变高**"✓ 不是"把内容区撑开"✓）
       *
       *   现在：普通模式 = 可用 × 0.5 ✓（用户要的那个"缩两次三成"✓）
       *        「更高」  = 可用 × 1.0 ✓（顶到屏幕边，但不顶出去 ✓）
       *   两者差一倍 ✓ 按下去一定看得见 ✓
       *   ⚠ 系数只在**这一处**算 ✗ 别在别处再乘一次 ✓（乘两回就是 0.25，矮得离谱 ✓）*/
      const _tall = sheet.classList.contains('mob-tall');
      /* 🔴 **系数按"页"分** ✗✗（2026-10-13 用户："但是颜色区的高度不要跟着调那么低啊"✓）
       *   前面那几个系数（0.7 → 0.5 → 0.333 → 0.4）都是冲着**设置页**调的 ✓
       *   可它是**全局**的 ✗ 于是**颜色页也跟着缩到四成** ✓
       *   颜色页里是调色板 + 图例列表 ✓ 本来就该占地方 ✓ 缩那么矮根本没法用 ✓
       *
       *   现在分开：
       *     · `display`（设置）→ **0.46** ✓
       *       🔴 **2026-10-17 +15%**（用户："手机版的高度再提升15%"✓
       *         原话还有一句："你的高度UI根本没搞定，敷衍我"✓ —— 骂得对 ✓
       *         ⚠ 我前两轮都在改 `mobile_css.txt` 里的 `#mob-sheet.mob-tall` ✗
       *           那是**白改**：真正的高度是下面那句内联 `sheet.style.maxHeight` ✓
       *           内联压过 CSS ✓（这个坑**旧注释里就写着**，我没读到 ✓ 我的错 ✓）
       *         `0.4 × 1.15 = 0.46` ✓
       *       （原来 0.4 是用户前几轮"缩两次三成"逐次调下来的值 ✓）
       *     · `color`（颜色）  → **1.0** ✓（回到缩放之前的高度 ✓）
       *     · 其余（工具 / 视图 / 文件那些）→ 也走 1.0 ✓
       *       （它们本来就短 ✓ 按内容撑就行 ✓ 缩它们没意义 ✓）
       *   ⚠ 判断用 `key` ✗ 它是 `place(btn, key)` 的入参 ✓ 现成的 ✓
       *     别去猜"当前哪页" ✗（`opened` 是别的作用域里的 ✓）*/
      const _isDisplay = (key === 'display');
      /* 🔴 设置页那个 0.46 = 0.4 × 1.15 ✓（2026-10-17 用户要的 +15% ✓
       *   改**这里**才是真改 ✗ —— CSS 里那两条被这句内联压过 ✓
       *
       * 🔴🔴 **2026-10-21 收敛：高度现在只有一个来源** ✓✓
       *
       *   原来**三套在抢**同一个高度：
       *     ① 这里：`_f` × 可用高度 → `sheet.style.maxHeight`
       *     ② 下面那句：`height = maxHeight`（「更高」时把高度定死）
       *     ③ `mobile_css.txt` 的 `#mob-sheet.mob-tall { height: 38vh !important }`
       *   ③ 那条 `!important` **把 ①② 一起压掉了** ✗
       *   → 「更高」无视屏幕、永远 38vh ✓ 小屏上太高、大屏上太矮 ✓
       *   （跟当年 `.tall` 那个死类是同一个病：**两个地方写同一个数** ✓）
       *
       *   → 删掉 ③ 的 `!important`（CSS 那边只留兜底 ✓ 见 mobile_css.txt）
       *     · 普通模式：**不写 height** → 面板按内容自己撑 ✓（矮、贴着按钮 ✓）
       *     · 「更高」  ：`height = maxHeight` → 撑到算出来的上限 ✓ 多的进滑条 ✓
       *   ⚠ 以后调高度**只改这一处** ✗ 别再往 CSS 里加大数字 ✓*/
      const _f = _tall ? 1.0 : (_isDisplay ? 0.46 : 1.0);
      const _cap = Math.round(Math.max(160, _avail) * _f) + 'px';
      if (below) {
        sheet.style.top = Math.round(r.bottom - a.top + 10) + 'px';
        sheet.style.bottom = 'auto';
        sheet.style.maxHeight = _cap;
      } else {
        sheet.style.bottom = Math.round(a.bottom - r.top + 10) + 'px';
        sheet.style.top = 'auto';
        sheet.style.maxHeight = _cap;
      }
      sheet.style.setProperty('--tail-x', Math.round(cx - left) + 'px');
      /* 「更高」模式：把 height 顶到**刚算出来的上限** ✓
       *   ⚠ 这里就是 **1.0** ✗ 别再乘一次 0.7 ✓ ——
       *     上面那个 `maxHeight` 已经按系数缩过了 ✓ 再乘一遍就矮得没法看 ✓
       *   （第一版就是两边都乘 ✗ 幸好发现得早 ✓）*/
      /* 「更高」模式：把 height 顶到**刚算出来的可用上限**（就是上面那个 max-height ✓
       *   贴着底部往上长 ✓ 不会顶出屏幕 ✓）；普通模式清掉 height ✓ 回到"按内容自己撑" ✓
       * 这里**不做动画** ✗（用户要求去掉 ✓ 而且显式 px 也会让面板不再跟内容长高 ✗）*/
      /* 「更高」模式：把 height 顶到**可用高度的七成**（用户 2026-10-13 定的："设置高度减去 30%"✓
       *   原来是直接 `height = maxHeight` ✗ —— 那是"从按钮一路顶到屏幕边"✓ 太大了 ✓
       *   而且它是**内联样式** ✗ 优先级压过 CSS ✓ 所以光改 `mobile_css.txt` 里的
       *   `#mob-sheet.mob-tall { height: 40vh }` **根本看不出变化** ✓
       *   （用户报的"根本没看到，高度仍然遵循界面最高的"就是这个 ✓）
       * 现在 ×0.7 ✓ 剩下的交给 `.mob-panes` 的滑条 ✓ 内容放得下 ✓
       * ⚠ `maxHeight` 是字符串（"512px" ✓）→ 要 `parseFloat` 再算 ✓ */
      try {
        if (sheet.classList.contains('mob-tall') && sheet.style.maxHeight) {
          /* ⚠ 1.0 ✓（`maxHeight` 上面已经乘过 0.7 了 ✗ 这儿再乘就成 0.49）*/
          sheet.style.height = sheet.style.maxHeight;
        } else {
          sheet.style.height = '';
        }
      } catch (e) { /* ✓ */ }
    } catch (e) { /* 算不出来就按默认位置 ✓ */ }
  };

  let opened = null, openedBtn = null;
  const setSheet = (key, btn) => {
    const g = key ? group[key] : null;
    for (const k in group) group[k].el.hidden = !(g && group[k] === g);
    if (g) {
      titleEl.textContent = g.title;
      /* 「更高」**按页各记各的** ✓ —— 打开哪一页就套哪一页的状态 ✓
       * ⚠ 这一句**必须在 place() 之前** ✗：
       *   ① 少了它，按钮加上的 mob-tall 就没人摘 ✗ → 之后打开工具/视图/文件也全变高 ✗
       *      （整个 App 只有一个 #mob-sheet ✓ 四个入口共用 ✓ 用户报过两次 ✓）
       *   ② place() 是**看着这个 class** 决定高度的 ✓ 顺序反了就会按上一页的状态算一次 ✗
       */
      try {
        const _ts = (group.display && group.display.el && group.display.el._tallState) || {};
        sheet.classList.toggle('mob-tall', !!_ts[key]);
      } catch (e) { /* ✓ */ }
      if (btn) place(btn, key);
      // 高度**不锁** ✓ 跟「颜色」那栏一样按内容自己撑 ✓（用户要求）—— 这一句是给别的面板的
      //（设置那一栏例外：下面用 _lockSoon 量出最高页锁住 ✓ 切页不跳 ✓）
      sheet.classList.remove('tall');
      if (key === 'display' && group.display && group.display.el._lockSoon) {
        requestAnimationFrame(() => { try { group.display.el._lockSoon(); } catch (e) { /* ✓ */ } });
      }
      // **横向滚动位置强制归零** ✓（不然内容稍宽就被推到一边、左边那截看不见 ✗）
      try { body.scrollLeft = 0; sheet.scrollLeft = 0; } catch (e) { /* ✓ */ }
      /* 打开设置面板时，要是当前正停在「导出图例」那一页，把列表重刷一遍 ✓
       * 用 legendFresh ✓ —— 它会**先按"现在这一刻的剧本/层级"把色块补算一遍**再列 ✓
       * （只刷列表的话，列出来的可能还是上一个剧本的分组 ✗ 用户报过 ✓）*/
      if (key === 'display') {
        const _dEl = group.display && group.display.el ? group.display.el : null;
        const _lgPane = _dEl ? _dEl.querySelector('.mob-pane[data-pane="legend"]') : null;
        if (_lgPane && !_lgPane.hidden) {
          try { legendFresh(); } catch (e) { /* ✓ */ }
        }
      }
      sheet.classList.add('on');
      backdrop.classList.add('on');
      opened = key;
      openedBtn = btn || null;
    } else {
      sheet.classList.remove('on');
      backdrop.classList.remove('on');
      opened = null;
      openedBtn = null;
    }
    const bs = ui.querySelectorAll('#mob-dock button');
    for (let i = 0; i < bs.length; i++) bs[i].classList.toggle('on', bs[i].getAttribute('data-sheet') === key);
    const io = $('mob-io');
    if (io) io.classList.toggle('on', key === 'io');
  };

  const dockWs = ui.querySelectorAll('#mob-dock button');
  for (let i = 0; i < dockWs.length; i++) {
    (function (b) {
      b.addEventListener('click', () => {
        const k = b.getAttribute('data-sheet');
        setSheet(opened === k ? null : k, b);
      });
    })(dockWs[i]);
  }
  const ioBtn = $('mob-io');
  if (ioBtn) ioBtn.addEventListener('click', () => setSheet(opened === 'io' ? null : 'io', ioBtn));
  const backBtn = $('mob-back');
  if (backBtn) backBtn.addEventListener('click', () => {
    // **只管一件事：回主菜单** ✓（用户定的 —— 不让它兼职关弹窗、收面板 ✗）
    // **而且不许把画的东西弄丢** ✓：先写进暂存区 ✓ 回来选同一张图会自动接上 ✓
    // （暂存是按地图分开存的 ✓ 换地图不会串味 ✓）
    let dirty = 0;
    try { dirty = (state.changed && state.changed.size) || 0; } catch (e) { dirty = 0; }
    if (dirty > 0 && !stashNow()) {
      // 存不进暂存区（无痕模式 / 配额满）才问一句 ✗ 至少不静默丢掉
      const ok = window.confirm('这台浏览器不让暂存，回主菜单会丢掉还没导出的涂色（改动 ' + dirty + ' 处）。\n'
        + '想留着就先点「文件 → 存入地图仓库」存一份。\n\n确定要回主菜单吗？');
      if (!ok) return;
    }
    // 换地图要把数据、渲染器整套重来 ✓ 页面自带的启动流程正好干这事 ✓
    // 重新加载之后，同一张图的涂色会由 loadProject() 自动接上 ✓
    location.reload();
  });
  if ($('mob-sheet-x')) $('mob-sheet-x').addEventListener('click', () => setSheet(null));
  if (backdrop) backdrop.addEventListener('click', () => setSheet(null));
  document.addEventListener('keydown', (e) => { if (e.key === 'Escape') setSheet(null); });
  // 横竖屏切换 / 转屏时，那一栏跟着按钮重新摆一下 ✓
  window.addEventListener('resize', () => { if (opened && openedBtn) place(openedBtn, opened); });

  /* ============================================================ 再搬节点 */
  const panelOf = (sel) => {
    const n = typeof sel === 'string' ? $(sel) : sel;
    return n && n.closest ? n.closest('section.panel') : null;
  };
  const move = (g, node) => { if (node && node.parentNode !== g) g.appendChild(node); };
  const movePanelOf = (g, sel) => { move(g, panelOf(sel)); };

  wait(() => {
    movePanelOf(gColor, 'brush-hex');        // 当前颜色：色块 / HEX / RGB / 标记
    movePanelOf(gColor, 'results');          // 搜索
    move(gColor, $('label-pick'));

    /* ============================================================ 颜色：取消二级菜单
     * 🔴 2026-10-27 用户：
     *   "关于二级窗口可以取消，调色板和常用颜色放在面板下方，
     *    那个颜色块的功能重新回调为改颜色。此外手机用不了右键，
     *    没法清除调色板的颜色很恼火"
     *
     *   原来点大色块弹 `#brush-menu`（自定义颜色 / 收藏 / 最近使用）✓
     *   手机上它是个浮层，还会被面板裁掉；而且用户点色块的本意就是**改颜色** ✓
     *   → 手机版直接取消这个二级窗口：
     *     ① 点大色块 = 打开系统取色器（走原来的 `#brush-menu-custom` 那条路 ✓）
     *     ② 收藏调色板 / 最近使用**搬到底部**，一直看得见 ✓
     *     ③ 色块上长按 = 删除（手机没有右键 ✓）
     *   ⚠ 只动手机版 ✓ 桌面版那颗色块仍然按 2026-10-24 的二级菜单走 ✓
     */
    const colorPanel = panelOf($('brush-swatch')) || gColor;
    const bmenu = $('brush-menu');
    const fav = $('palette-fav');
    const recent = $('palette');
    if (colorPanel && bmenu && fav && recent) {
      /* 把二级菜单里那两段（标题 + 色块）整段搬出来 ✓
         ⚠ 标题节点是 palette 的前一个兄弟 ✗ 别只搬 palette 色块 ✓
           （不然面板底下会冒出两排没名字的方块 ✓）*/
      const kids = Array.prototype.slice.call(bmenu.children || []);
      const favCap = kids[kids.indexOf(fav) - 1] || null;
      const recentCap = kids[kids.indexOf(recent) - 1] || null;
      const kit = document.createElement('div');
      kit.className = 'mob-color-kit';
      if (favCap) kit.appendChild(favCap);
      kit.appendChild(fav);
      if (recentCap) kit.appendChild(recentCap);
      kit.appendChild(recent);
      /* 手机没有右键，长按是个隐藏手势 ✗ 给一行小字说清楚 ✓ */
      const hint = document.createElement('div');
      hint.className = 'mob-color-hint';
      hint.textContent = '长按色块可以删掉';
      kit.appendChild(hint);
      colorPanel.appendChild(kit);
      /* 二级菜单整个退役 ✓（自定义颜色那个按钮还留着，只是不显示 ✓
         大色块的 onclick 会直接调它 ✓）*/
      bmenu.hidden = true;
      bmenu.style.display = 'none';
    }

     *   原因：缩放本身是**合成 wheel 事件**交给上层去做的 ✓ 而那一层也会改 cam ✓

     *   → 所以缩放这块**保持增量** ✓ 想让它稳只能从别处下手 ✓
     *   唯一保留的改动：阈值 0.002 → 0.01 ✓

    const mid = $('mob-mid');
    const undo = $('btn-undo');
    const redo = $('btn-redo');
    if (!mid) return;
    const box = document.createElement('span');
    box.className = 'mob-undo';
    mid.appendChild(box);
    if (undo) { undo.classList.remove('btn', 'ghost'); undo.classList.add('mob-icobtn'); box.appendChild(undo); }
    if (redo) { redo.classList.remove('btn', 'ghost'); redo.classList.add('mob-icobtn'); box.appendChild(redo); }

  /* ============================================================ 教程：手机版改成"一个介绍弹窗"
   * 逐步引导在手机上指不准 ✗（它要圈顶栏那些按钮，可手机端都搬进面板了 ✓）
   * 所以：点「教程」= 弹一个介绍窗 ✓ 文案也是介绍性质的 ✓ 不再让人一步步照做 ✓
   * 电脑版那份逐步教程是另一套 ✓（`web/js/tutorial.js` ✓）
   *
   * 🔴 **2026-10-20 整篇重写** ✗✗（用户："手机版和电脑版教程重新做一下，过时不知道多久了"）
   *   过时的地方：
   *     · 说「底下四个按钮」✗ —— 现在是**五个**（颜色 / 工具 / 视图 / 设置 / 文件 ✓）
   *       而且原来把「文件」漏了，用户按文案会找不着导出在哪 ✓
   *     · 说「视图：换年份剧本，或者切到地区、省份」✗
   *       v1.10 之后视图页里是**两级**：上面「剧本」折叠栏（点开才是年份），
   *       下面才是**地理层级**（大洲 / 大区 / 区域 / 地区 / 省份 ✓）
   *     · 没提「折叠」这件事（手机版那条顶栏能收起来 ✓）

  const introHTML =

    + '<strong>工具</strong>：涂抹、填色、橡皮、改名、吸管、查看，一共六个。<br>'
    + '　　涂抹是按住拖动一片；<strong>填色</strong>是点一下变化一次（担心手滑就用它）；橡皮则是擦回原样。<br>'
    + '<strong>视图</strong>：分成两级。上面那个<strong>「剧本」折叠栏</strong>，点开之后才是年份'

   * 只在手机补丁里 ✓ 电脑版照旧有高亮 ✓

  try {
    const _clearHl = () => {
      try {
        if (typeof renderer !== 'undefined' && renderer && renderer.setHover) {
          renderer.setHover(null);
        }
        if (typeof state !== 'undefined' && state && state.hover) {
          state.hover.pid = 0;
          state.hover.inside = false;
        }
        /* ⚠⚠ **"整片编组高亮"还得单独清一次** ✗✗（2026-10-13 用户报"手机版为什么还有高亮"）
         *   `setHoverState` 一共 **6 个参数**，最后那个是**编组号** ✓

        if (renderer && renderer.setHoverState) renderer.setHoverState(0, 0, null, 0, null, 0);
      } catch (e) { /* 不影响用 ✓ */ }
    };
    for (const _ev of ['pointerdown', 'pointermove', 'pointerup', 'pointercancel',
                       'touchstart', 'touchmove', 'touchend', 'wheel']) {
      document.addEventListener(_ev, _clearHl, { capture: true, passive: true });
    }
    // 一开始也清一次 ✓（免得进页面就挂着一圈）
    _clearHl();
  } catch (e) { /* ✓ */ }
}

// ================================================================ 手机端：点地块不高亮
// 触屏没有"把鼠标移开"这个动作 —— 手指点一下，高亮就黏在屏幕上消不掉 ✗
// 看着像"这块被选中了"，所以手机端**任何情况下都不点亮**（信息卡照常出 ✓）
{
  const _hlDesk = applyHoverHighlight;
  applyHoverHighlight = function (pid) {
    _hlDesk(pid || 0, null);     // 状态照旧更新：卡片 / 光标逻辑还要用 ✓
    renderer.hoverPid = 0;       // 单块高亮：关 ✓
    renderer.hoverPaintOn = 0;   // 整族高亮（同色一族一起亮）：也关 ✓
    renderer.setHover(null);
    /* 🔴 **整片编组高亮：这里才是漏掉的那一路** ✗✗
     *   （2026-10-13 用户报："手机版**为什么还有高亮**、、、"✓ 语气很有画面感 ✓）
     *   `setHoverState(pid, paintOn, rgb, ?, ?, group)` 一共 **6 个参数** ✓
     *   最后那个 `group` 是**编组号** ✓
     *   （`hoverGroupIdOf(pid)` 算出来的 ✓ 见 gl.js 的 rebuildHoverGroups ✓）
     *   ⚠ 上面三句**都碰不到它** ✗：
     *     · `renderer.hoverPid = 0`      → 只管单块 ✓
     *     · `renderer.hoverPaintOn = 0`  → 只管同色整族 ✓
     *     · `renderer.setHover(null)`    → 也只管前五个 ✓
     *   → 这里再推一次全零 ✓ 六个数一个不留 ✓ */
    if (renderer.setHoverState) renderer.setHoverState(0, 0, null, 0, null, 0);
  };
}
